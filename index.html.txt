function switchPage(pageId){
  playClick();
  pages.forEach(p=>p.classList.remove("active"));
  navItems.forEach(n=>n.classList.remove("active"));
  document.getElementById(pageId).classList.add("active");
  refreshMyStat();
}
function createParticle(e){
  const x = e.clientX;
  const y = e.clientY;
  const colors = ['#7cf28b','#38bdf8','#a855f7','#ff6b72'];
  for(let i=0;i<8;i++){
    const d = document.createElement("div");
    d.className="particle";
    const tx = (Math.random()-0.5)*120;
    const ty = (Math.random()-0.5)*120;
    d.style.setProperty('--tx',tx+"px");
    d.style.setProperty('--ty',ty+"px");
    d.style.left = x+"px";
    d.style.top = y+"px";
    d.style.background = colors[Math.floor(Math.random()*colors.length)];
    document.getElementById("particles").appendChild(d);
    setTimeout(()=>d.remove(),900);
  }
}

avatarFile.addEventListener("change",(e)=>{
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = ev=>{
    saveData.avatar = ev.target.result;
    avatarImg.src = saveData.avatar;
    saveAll();
  }
  reader.readAsDataURL(file);
  playNotify();
})

function createNewOrder(projectName){
  const uname = saveData.username || "匿名下单用户";
  const price = PRICE_MAP[projectName];
  const newOrder = {
    orderId:Date.now(),
    userName:uname,
    projectName:projectName,
    price:price
  }
  saveData.orderList.push(newOrder);
  saveAll();
  refreshMyStat();
  playNotify();
  openModal(orderTipModal);
  alert(`下单成功！订单单号：${newOrder.orderId}｜项目：${projectName}｜金额：${price.toFixed(2)}元`);
}

function refreshMyStat(){
  const meName = saveData.username || "匿名下单用户";
  const myOrders = saveData.orderList.filter(o=>o.userName===meName);
  const count = myOrders.length;
  const total = myOrders.reduce((sum,o)=>sum+o.price,0);
  myOrderCountDom.innerText = count;
  myTotalMoneyDom.innerText = total.toFixed(2);
}
